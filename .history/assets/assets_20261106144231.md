const $ = document.querySelector.bind(document);
const $$ = document.querySelectorAll.bind(document);

/**
 * Hàm tải template
 *
 * Cách dùng:
 * <div id="parent"></div>
 * <script>
 *  load("#parent", "./path-to-template.html");
 * </script>
 */
// Trong assets/js/script.js (Dòng 13 - 29)
function load(selector, path) {
    const targetEl = $(selector);
    if (!targetEl) {
        console.warn(`[load] Không tìm thấy thẻ "${selector}" trong index.html!`);
        return;
    }

    const cached = localStorage.getItem(path);
    if (cached && cached.trim() !== "") {
        targetEl.innerHTML = cached;
    }

    fetch(path)
        .then((res) => {
            if (!res.ok) {
                throw new Error(`HTTP ${res.status} - Không tìm thấy file tại đường dẫn: ${path}`);
            }
            return res.text();
        })
        .then((html) => {
            if (html.trim() === "") {
                console.warn(`[load] File "${path}" đang RỖNG (0 ký tự). Hãy kiểm tra và bấm Ctrl+S trong VS Code!`);
                return;
            }
            if (html !== cached) {
                targetEl.innerHTML = html;
                localStorage.setItem(path, html);
            }
        })
        .catch((err) => {
            console.error(`[load] Lỗi khi nạp "${path}":`, err.message);
        })
        .finally(() => {
            window.dispatchEvent(new Event("template-loaded"));
        });
}

/**
 * Hàm kiểm tra một phần tử
 * có bị ẩn bởi display: none không
 */
function isHidden(element) {
    if (!element) return true;

    if (window.getComputedStyle(element).display === "none") {
        return true;
    }

    let parent = element.parentElement;
    while (parent) {
        if (window.getComputedStyle(parent).display === "none") {
            return true;
        }
        parent = parent.parentElement;
    }

    return false;
}

/**
 * Hàm buộc một hành động phải đợi
 * sau một khoảng thời gian mới được thực thi
 */
function debounce(func, timeout = 300) {
    let timer;
    return (...args) => {
        clearTimeout(timer);
        timer = setTimeout(() => {
            func.apply(this, args);
        }, timeout);
    };
}

/**
 * Hàm tính toán vị trí arrow cho dropdown
 *
 * Cách dùng:
 * 1. Thêm class "js-dropdown-list" vào thẻ ul cấp 1
 * 2. CSS "left" cho arrow qua biến "--arrow-left-pos"
 */
const calArrowPos = debounce(() => {
    if (isHidden($(".js-dropdown-list"))) return;

    const items = $$(".js-dropdown-list > li");

    items.forEach((item) => {
        const arrowPos = item.offsetLeft + item.offsetWidth / 2;
        item.style.setProperty("--arrow-left-pos", `${arrowPos}px`);
    });
});

window.addEventListener("template-loaded", calArrowPos);

/**
 * Giữ active menu khi hover
 *
 * Cách dùng:
 * 1. Thêm class "js-menu-list" vào thẻ ul menu chính
 * 2. Thêm class "js-dropdown" vào class "dropdown" hiện tại
 *  nếu muốn reset lại item active khi ẩn menu
 */
window.addEventListener("template-loaded", handleActiveMenu);

function handleActiveMenu() {
    const dropdowns = $$(".js-dropdown");
    const menus = $$(".js-menu-list");
    const activeClass = "menu-column__item--active";

    const removeActive = (menu) => {
        menu.querySelector(`.${activeClass}`)?.classList.remove(activeClass);
    };

    const init = () => {
        menus.forEach((menu) => {
            const items = menu.children;
            if (!items.length) return;

            removeActive(menu);
            if (window.innerWidth > 991) items[0].classList.add(activeClass);

            Array.from(items).forEach((item) => {
                item.onmouseenter = () => {
                    if (window.innerWidth <= 991) return;
                    removeActive(menu);
                    item.classList.add(activeClass);
                };
                item.onclick = () => {
                    if (window.innerWidth > 991) return;
                    removeActive(menu);
                    item.classList.add(activeClass);
                    item.scrollIntoView();
                };
            });
        });
    };

    init();

    dropdowns.forEach((dropdown) => {
        dropdown.onmouseleave = () => init();
    });
}

/**
 * JS toggle
 *
 * Cách dùng:
 * <button class="js-toggle" toggle-target="#box">Click</button>
 * <div id="box">Content show/hide</div>
 */
window.addEventListener("template-loaded", initJsToggle);

function initJsToggle() {
    $$(".js-toggle").forEach((button) => {
        // Đọc qua dataset.toggleTarget chuẩn HTML5 (hỗ trợ lùi về toggle-target cũ nếu còn sót)
        const target = button.dataset.toggleTarget || button.getAttribute("toggle-target");
        if (!target) return;

        button.onclick = (e) => {
            e.preventDefault();
            const targetEl = $(target);
            if (!targetEl) return;

            const isHidden = targetEl.classList.contains("hide");
            requestAnimationFrame(() => {
                targetEl.classList.toggle("hide", !isHidden);
                targetEl.classList.toggle("show", isHidden);
                button.setAttribute("aria-expanded", String(isHidden));
            });
        };
    });
}

window.addEventListener("template-loaded", () => {
    const links = $$(".js-dropdown-list > li > a");

    links.forEach((link) => {
        link.onclick = () => {
            if (window.innerWidth > 991) return;
            const item = link.closest("li");
            item.classList.toggle("navbar__item--active");
        };
    });
});

// Chuyên trách duy nhất templates/header.html: Sticky Scroll, Đồng hồ thời gian thực, Dropdown Header & Theme
(() => {

    // Khôi phục Theme Ngũ hành ngay lập tức

        document.querySelectorAll(".header__theme-btn").forEach((btn) => {
            const isMatch = btn.dataset.theme === selectedTheme;
            btn.classList.toggle("is-active", isMatch);
            btn.setAttribute("aria-pressed", String(Boolean(isMatch)));
        });

        // Phát sự kiện để footer.js hoặc các component khác tự đồng bộ trạng thái
        window.dispatchEvent(new CustomEvent("theme-changed", { detail: { theme: selectedTheme } }));
    };

    // Xuất hàm đổi theme toàn cục để footer.js gọi khi cần
    window.setFengShuiTheme = applyTheme;

    const updateHeaderClock = () => {
        const dateEl = document.getElementById("current-date");
        const timeEl = document.getElementById("current-time");
        if (!dateEl || !timeEl) return;

        const now = new Date();
        const pad = (n) => String(n).padStart(2, "0");

        const dd = pad(now.getDate());
        const mm = pad(now.getMonth() + 1);
        const yyyy = now.getFullYear();
        const hh = pad(now.getHours());
        const min = pad(now.getMinutes());
        const ss = pad(now.getSeconds());

        dateEl.textContent = `${dd}/${mm}/${yyyy}`;
        dateEl.setAttribute("datetime", `${yyyy}-${mm}-${dd}`);
        timeEl.textContent = `${hh}:${min}:${ss}`;
        timeEl.setAttribute("datetime", `${hh}:${min}:${ss}`);
    };

    const initHeader = () => {
        const header = document.getElementById("header") || document.getElementById("site-header");
        if (!header) return;

        const activeTheme = document.documentElement.getAttribute("data-theme") || "earth";
        const widgetBtn = document.getElementById("widget-trigger-btn");
        const dropdowns = header.querySelectorAll(".js-header-dropdown");
        const themeBtns = header.querySelectorAll(".header__theme-btn");

        header.classList.toggle("header--scrolled", window.scrollY > 10);

        if (widgetBtn) {
            widgetBtn.onclick = () => {
                const isExpanded = widgetBtn.getAttribute("aria-expanded") === "true";
                const nextState = !isExpanded;
                widgetBtn.classList.toggle("is-active", nextState);
                widgetBtn.setAttribute("aria-expanded", String(nextState));
            };
        }

        dropdowns.forEach((current) => {
            current.ontoggle = () => {
                if (current.open) {
                    document.querySelectorAll(".js-header-dropdown[open], .js-cal-dropdown[open]").forEach((other) => {
                        if (other !== current) {
                            other.removeAttribute("open");
                        }
                    });
                }
            };
        });

        themeBtns.forEach((btn) => {
            const isCurrent = btn.dataset.theme === activeTheme;
            btn.classList.toggle("is-active", isCurrent);
            btn.setAttribute("aria-pressed", String(isCurrent));

            btn.onclick = () => {
                applyTheme(btn.dataset.theme);
                btn.closest("details")?.removeAttribute("open");
            };
        });

        updateHeaderClock();
        if (clockTimer) clearInterval(clockTimer);
        clockTimer = setInterval(updateHeaderClock, 1000);
    };

    const bindHeaderGlobalListeners = () => {
        if (isHeaderGlobalBound) return;
        isHeaderGlobalBound = true;

        window.addEventListener(
            "scroll",
            () => {
                const header = document.getElementById("header") || document.getElementById("site-header");
                if (header) {
                    header.classList.toggle("header--scrolled", window.scrollY > 10);
                }
            },
            { passive: true },
        );

        document.addEventListener("click", (e) => {
            document.querySelectorAll(".js-header-dropdown[open], .js-cal-dropdown[open]").forEach((dropdown) => {
                if (!dropdown.contains(e.target)) {
                    dropdown.removeAttribute("open");
                }
            });
        });

        document.addEventListener("keydown", (e) => {
            if (e.key === "Escape") {
                document.querySelectorAll(".js-header-dropdown[open], .js-cal-dropdown[open]").forEach((dropdown) => {
                    dropdown.removeAttribute("open");
                });
            }
        });
    };

    document.addEventListener("DOMContentLoaded", () => {
        bindHeaderGlobalListeners();
        initHeader();
    });

    window.addEventListener("template-loaded", initHeader);
})();

// Chuyên trách duy nhất templates/sidebar.html: Ngăn kéo Mobile Drawer, Khóa cuộn Body & Active Link
(() => {
    let isSidebarGlobalBound = false;

    const toggleSidebar = (forceClose = false) => {
        const sidebar = document.getElementById("calendar-sidebar");
        const burgerBtn = document.getElementById("burger-btn");
        const backdrop = document.getElementById("sidebar-backdrop");
        if (!sidebar || !burgerBtn || !backdrop) return;

        const isOpen = sidebar.classList.contains("is-open");
        const nextState = forceClose ? false : !isOpen;

        sidebar.classList.toggle("is-open", nextState);
        burgerBtn.classList.toggle("is-active", nextState);
        backdrop.classList.toggle("is-active", nextState);
        burgerBtn.setAttribute("aria-expanded", String(nextState));

        // Khóa cuộn trang trên <body> khi mở menu Mobile
        document.body.style.overflow = nextState ? "hidden" : "";
    };

    const syncActiveSidebarLink = () => {
        const currentPage = window.location.pathname.split("/").pop() || "index.html";
        const sidebarLinks = document.querySelectorAll(".sidebar__link[data-nav-page]");

        sidebarLinks.forEach((link) => {
            const isMatch = link.dataset.navPage === currentPage;
            link.classList.toggle("sidebar__link--active", isMatch);
            if (isMatch) {
                link.setAttribute("aria-current", "page");
            } else {
                link.removeAttribute("aria-current");
            }
        });
    };

    const initSidebar = () => {
        const burgerBtn = document.getElementById("burger-btn");
        const backdrop = document.getElementById("sidebar-backdrop");

        if (burgerBtn) burgerBtn.onclick = () => toggleSidebar();
        if (backdrop) backdrop.onclick = () => toggleSidebar(true);

        syncActiveSidebarLink();
    };

    const bindSidebarGlobalListeners = () => {
        if (isSidebarGlobalBound) return;
        isSidebarGlobalBound = true;

        window.addEventListener("resize", () => {
            const sidebar = document.getElementById("calendar-sidebar");
            if (window.innerWidth >= 768 && sidebar?.classList.contains("is-open")) {
                toggleSidebar(true);
            }
        });

        document.addEventListener("keydown", (e) => {
            if (e.key === "Escape") {
                toggleSidebar(true);
            }
        });
    };

    document.addEventListener("DOMContentLoaded", () => {
        bindSidebarGlobalListeners();
        initSidebar();
    });

    window.addEventListener("template-loaded", initSidebar);
})();

        if (e.detail?.theme) syncFooterThemeDots(e.detail.theme);
    });

    document.addEventListener("DOMContentLoaded", initFooter);
    window.addEventListener("template-loaded", initFooter);
})();

// Chuyên trách duy nhất Trang Đồng hồ (clock.html): Đồng hồ TimeSpot, Bát Tự Tứ Trụ, 12 Canh Giờ & Thời tiết
(() => {
    let clockInterval = null;
    let lastRenderedBranchIndex = -1;
    let selectedBranchIndex = -1;

    const getZonedDateParts = (timeZone) => {
        const now = new Date();
        const formatter = new Intl.DateTimeFormat("en-US", {
            timeZone,
            year: "numeric",
            month: "2-digit",
            day: "2-digit",
            hour: "2-digit",
            minute: "2-digit",
            second: "2-digit",
            hour12: false,
        });
        const parts = Object.fromEntries(formatter.formatToParts(now).map((p) => [p.type, p.value]));
        return {
            year: Number(parts.year),
            month: Number(parts.month),
            day: Number(parts.day),
            hour: Number(parts.hour) % 24,
            minute: Number(parts.minute),
            second: Number(parts.second),
        };
    };

    const updateSelectedHourDetail = (hourObj, solarTerm) => {
        if (!hourObj) return;
        const nameEl = document.getElementById("detail-selected-hour-name");
        const napamEl = document.getElementById("detail-selected-hour-napam");
        const termEl = document.getElementById("detail-selected-solarterm");

        if (nameEl) nameEl.textContent = `Giờ ${hourObj.canchi} (${hourObj.range})`;
        if (napamEl) napamEl.textContent = hourObj.napAm;
        if (termEl) termEl.textContent = solarTerm;
    };

    const renderZodiacGrid = (dayInfo) => {
        const grid = document.getElementById("clock-zodiac-grid");
        if (!grid) return;

        const activeFilter =
            document.querySelector(".clock-zodiac-section__filter-btn.is-active")?.dataset.hourFilter || "all";

        grid.innerHTML = dayInfo.hours12
            .map((h) => {
                const isCurrent = h.index === dayInfo.currentBranchIdx;
                const isSelected = h.index === selectedBranchIndex;
                const fortuneKey = h.isGood ? "good" : "bad";
                const fortuneLabel = h.isGood ? "Hoàng Đạo" : "Hắc Đạo";
                const fortuneIcon = h.isGood ? "☀️" : "🌙";

                selectedBranchIndex = Number(card.dataset.branchIndex);

                renderZodiacGrid(dayInfo);

            card.onkeydown = (e) => {

                if (e.key === "ArrowLeft" || e.key === "ArrowUp") nextIdx = (idx - 1 + cards.length) % cards.length;

                if (nextIdx !== -1) {
                    e.preventDefault();
                    selectedBranchIndex = Number(cards[nextIdx].dataset.branchIndex);
                    updateSelectedHourDetail(dayInfo.hours12[selectedBranchIndex], dayInfo.solarTerm);
                    renderZodiacGrid(dayInfo);
                    grid.querySelector(`[data-branch-index="${selectedBranchIndex}"]`)?.focus();
                }
            };
        });
    };

    const fetchWeatherAndSun = async (lat, lon, tz) => {
        try {
            const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current_weather=true&daily=sunrise,sunset&timezone=${encodeURIComponent(tz)}`;
            const res = await fetch(url);

                const [rH, rM] = rise.split(":").map(Number);
                const [sH, sM] = set.split(":").map(Number);
                const totalMins = sH * 60 + sM - (rH * 60 + rM);
                const dH = Math.floor(totalMins / 60);
                const dM = totalMins % 60;
                const sunEl = document.getElementById("clock-sun-times") || document.getElementById("clock-sun- times");
                if (sunEl) {
                    sunEl.textContent = `${rise} - ${set} (${dH}h ${String(dM).padStart(2, "0")}m)`;
                }
            }

            // Giữ giá trị mặc định nếu không có kết nối mạng
        }
    };

    const initClockPage = () => {
        const rootEl = document.getElementById("clock-calendar-root");
        const app = document.getElementById("clock-bento-app");
        if (!app || !window.LunarEngine) return;

        if (rootEl) rootEl.setAttribute("aria-busy", "false");
        if (app.dataset.bound === "true") return;
        app.dataset.bound = "true";

        const tickClock = () => {
            const tz = app.dataset.timezone || "Asia/Ho_Chi_Minh";
            const format = app.dataset.activeFormat || "24";
            const parts = getZonedDateParts(tz);
            const pad = (n) => String(n).padStart(2, "0");

            let displayHour = parts.hour;
            const ampmEl = document.getElementById("hero-clock-ampm");

            if (format === "12") {
                const suffix = parts.hour >= 12 ? "PM" : "AM";
                displayHour = parts.hour % 12 || 12;
                if (ampmEl) {
                    ampmEl.textContent = suffix;
                    ampmEl.hidden = false;
                }

            }

            const secondsEl = app.querySelector('[data-clock-part="seconds"]');

            if (minutesEl) minutesEl.textContent = pad(parts.minute);
            if (secondsEl) secondsEl.textContent = pad(parts.second);

                if (statusPill && currentHourInfo) {
                    statusPill.dataset.fortune = currentHourInfo.isGood ? "good" : "bad";

        const prevMonth = currentMonthNumber === 1 ? 12 : currentMonthNumber - 1;
        const prevYear = currentMonthNumber === 1 ? currentYearNumber - 1 : currentYearNumber;

                cellYear = prevYear;
                isMuted = true;
            } else if (i < leadingMutedCount + daysInCurrentMonth) {
                cellDay = i - leadingMutedCount + 1;
                cellMonth = currentMonthNumber;
                cellYear = currentYearNumber;
            } else {
                cellDay = i - (leadingMutedCount + daysInCurrentMonth) + 1;

                cellYear = nextYear;
                isMuted = true;

            const isActive = !isMuted && cellDay === targetActiveDay;

            const statusTokens = [];

            if (dayInfo.isHoliday) statusTokens.push("holiday");
            if (isToday) statusTokens.push("today");

            if (dayInfo.isHoliday) {
                dotsInnerHtml += `<span class="month-cal__dot month-cal__dot--holiday"></span>`;

            generatedCells.push(btn);
            fragment.appendChild(btn);
        }

        grid.innerHTML = "";
        grid.appendChild(fragment);

        if (activeCellRef) syncCellToBlocCard(activeCellRef);

        const grid = document.getElementById("month-cal-grid");
        if (!grid || grid.dataset.bound === "true") return;
        grid.dataset.bound = "true";

        if (nextMonthBtn) {
            nextMonthBtn.onclick = () => {
                const targetMonth = currentMonthNumber === 12 ? 1 : currentMonthNumber + 1;
                const targetYear = currentMonthNumber === 12 ? currentYearNumber + 1 : currentYearNumber;
                renderMonthCalendarGrid(targetYear, targetMonth, null);

            };
        }

            sameDayBtn.onclick = (e) => {

                e.stopPropagation();
                if (!activeDayIntlEvents.length) return;

                currentIntlEventIndex = (currentIntlEventIndex + 1) % activeDayIntlEvents.length;
                intlBtn.classList.add("is-active");
                sameDayBtn?.classList.remove("is-active");

                const nextYear = opt.dataset.year ? Number(opt.dataset.year) : currentYearNumber;

                current.removeAttribute("open");
                renderMonthCalendarGrid(nextYear, nextMonth, null);
            };
        });

        const highlightCellsByStatus = (statusKey) => {
            grid.querySelectorAll(".month-cal__cell").forEach((cell) => {
                if (!statusKey) {
                    cell.classList.remove("is-dimmed", "is-highlighted");
                    return;
                }
                const cellStatus = cell.dataset.status || "";
                const isMatch = cellStatus.split(" ").includes(statusKey);
                cell.classList.toggle("is-highlighted", isMatch);
                cell.classList.toggle("is-dimmed", !isMatch);
            });
        };

        legendPills.forEach((pill) => {
            const filterKey = pill.dataset.filter;
            pill.onmouseenter = () => highlightCellsByStatus(filterKey);
            pill.onmouseleave = () => highlightCellsByStatus(null);
            pill.onfocus = () => highlightCellsByStatus(filterKey);
            pill.onblur = () => highlightCellsByStatus(null);
        });

        // ĐÃ ĐẶT ĐÚNG VỊ TRÍ BÊN TRONG initMonthCalendar():
        window.addEventListener("converter-jump-to-date", (e) => {
            if (e.detail) {
                renderMonthCalendarGrid(e.detail.year, e.detail.month, e.detail.day);
            }
        });

        const urlDateParam = new URLSearchParams(window.location.search).get("date");
        if (urlDateParam && /^\d{4}-\d{2}-\d{2}$/.test(urlDateParam)) {
            const [y, m, d] = urlDateParam.split("-").map(Number);
            renderMonthCalendarGrid(y, m, d);
        } else {
            renderMonthCalendarGrid(currentYearNumber, currentMonthNumber, selectedSolarDay);
        }
    };

    document.addEventListener("DOMContentLoaded", initMonthCalendar);
})();

                // Nếu đang ở Trang chủ (có hàm nhảy ngày hoặc lưới lịch tháng), phát sự kiện nhảy thẳng tới ngày đó
                if (document.getElementById("month-cal-grid")) {
                    window.dispatchEvent(
                        new CustomEvent("converter-jump-to-date", {
                            detail: { year, month, day },
                        }),
                    );
                    document.getElementById("bloc-media-box")?.scrollIntoView({ behavior: "smooth" });
                } else {
                    window.location.href = `./index.html?date=${year}-${pad(month)}-${pad(day)}`;
                }
            };
        }

        setConverterToToday();
    };

    if (!isModalGlobalBound) {
        isModalGlobalBound = true;
        document.addEventListener("keydown", (e) => {
            const modal = document.getElementById("converter-modal");
            if (e.key === "Escape" && modal && !modal.hidden) {
                toggleConverterModal(false);
            }
        });
    }

    document.addEventListener("DOMContentLoaded", initConverterModal);
    window.addEventListener("template-loaded", initConverterModal);
})();
