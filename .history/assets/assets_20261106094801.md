// assets/js/header.js
// Chuyên trách Layout dùng chung: Sticky Header, Đồng hồ thời gian thực, Theme Ngũ hành, Sidebar Mobile & Footer
(() => {
    const THEME_STORAGE_KEY = "feng_shui_theme";
    let isGlobalBound = false;
    let clockTimer = null;

    // 1. Khôi phục Theme Ngũ hành từ localStorage ngay khi nạp script
    const savedTheme = localStorage.getItem(THEME_STORAGE_KEY) || "earth";
    document.documentElement.setAttribute("data-theme", savedTheme);

    const applyTheme = (selectedTheme) => {
        if (!selectedTheme) return;
        document.documentElement.setAttribute("data-theme", selectedTheme);
        localStorage.setItem(THEME_STORAGE_KEY, selectedTheme);

        document.querySelectorAll(".header__theme-btn").forEach((btn) => {
            const isMatch = btn.dataset.theme === selectedTheme;
            btn.classList.toggle("is-active", isMatch);
            btn.setAttribute("aria-pressed", String(Boolean(isMatch)));
        });

        document.querySelectorAll(".footer__element-dot").forEach((dot) => {
            dot.classList.toggle("is-active", dot.dataset.quickTheme === selectedTheme);
        });
    };

    // 2. Bật/tắt Sidebar Drawer trên Mobile & Khóa cuộn trang (Body Scroll Lock)
    const toggleSidebar = (forceClose = false) => {
        const sidebar = document.getElementById("calendar-sidebar");
        const burgerBtn = document.getElementById("burger-btn");
        const backdrop = document.getElementById("sidebar-backdrop");
        if (!sidebar || !burgerBtn || !backdrop) return;

        const isOpen = sidebar.classList.contains("is-open");
        const nextState = forceClose ? false : !isOpen;

        sidebar.classList.toggle("is-open", nextState);
        burgerBtn.classList.toggle("is-active", nextState);
        backdrop.classList.toggle("is-active", nextState);
        burgerBtn.setAttribute("aria-expanded", String(nextState));

        document.body.style.overflow = nextState ? "hidden" : "";
    };

    // 3. Đồng bộ trạng thái Active trên thanh điều hướng Sidebar theo trang hiện tại
    const syncActiveSidebarLink = () => {
        const currentPage = window.location.pathname.split("/").pop() || "index.html";
        const sidebarLinks = document.querySelectorAll(".sidebar__link[data-nav-page]");

        sidebarLinks.forEach((link) => {
            const isMatch = link.dataset.navPage === currentPage;
            link.classList.toggle("sidebar__link--active", isMatch);
            if (isMatch) {
                link.setAttribute("aria-current", "page");
            } else {
                link.removeAttribute("aria-current");
            }
        });
    };

    // 4. Cập nhật Ngày & Giờ thực trên Header
    const updateClock = () => {
        const dateEl = document.getElementById("current-date");
        const timeEl = document.getElementById("current-time");
        if (!dateEl || !timeEl) return;

        const now = new Date();
        const pad = (n) => String(n).padStart(2, "0");

        const dd = pad(now.getDate());
        const mm = pad(now.getMonth() + 1);
        const yyyy = now.getFullYear();
        const hh = pad(now.getHours());
        const min = pad(now.getMinutes());
        const ss = pad(now.getSeconds());

        dateEl.textContent = `${dd}/${mm}/${yyyy}`;
        dateEl.setAttribute("datetime", `${yyyy}-${mm}-${dd}`);
        timeEl.textContent = `${hh}:${min}:${ss}`;
        timeEl.setAttribute("datetime", `${hh}:${min}:${ss}`);
    };

    // 5. Khởi tạo Header, Sidebar & Footer
    const initHeaderAndSidebar = () => {
        const header = document.getElementById("header") || document.getElementById("site-header");
        const activeTheme = document.documentElement.getAttribute("data-theme") || "earth";

        if (header) {
            const burgerBtn = document.getElementById("burger-btn");
            const backdrop = document.getElementById("sidebar-backdrop");
            const widgetBtn = document.getElementById("widget-trigger-btn");
            const dropdowns = document.querySelectorAll(".js-header-dropdown");
            const themeBtns = document.querySelectorAll(".header__theme-btn");

            header.classList.toggle("header--scrolled", window.scrollY > 10);

            if (burgerBtn) burgerBtn.onclick = () => toggleSidebar();
            if (backdrop) backdrop.onclick = () => toggleSidebar(true);

            if (widgetBtn) {
                widgetBtn.onclick = () => {
                    const isExpanded = widgetBtn.getAttribute("aria-expanded") === "true";
                    const nextState = !isExpanded;
                    widgetBtn.classList.toggle("is-active", nextState);
                    widgetBtn.setAttribute("aria-expanded", String(nextState));
                };
            }

            dropdowns.forEach((current) => {
                current.ontoggle = () => {
                    if (current.open) {
                        document
                            .querySelectorAll(".js-header-dropdown[open], .js-cal-dropdown[open]")
                            .forEach((other) => {
                                if (other !== current) {
                                    other.removeAttribute("open");
                                }
                            });
                    }
                };
            });

            themeBtns.forEach((btn) => {
                const isCurrent = btn.dataset.theme === activeTheme;
                btn.classList.toggle("is-active", isCurrent);
                btn.setAttribute("aria-pressed", String(isCurrent));

                btn.onclick = () => {
                    applyTheme(btn.dataset.theme);
                    btn.closest("details")?.removeAttribute("open");
                };
            });

            updateClock();
            if (clockTimer) clearInterval(clockTimer);
            clockTimer = setInterval(updateClock, 1000);
        }

        document.querySelectorAll(".footer__element-dot").forEach((dot) => {
            dot.classList.toggle("is-active", dot.dataset.quickTheme === activeTheme);
            dot.onclick = () => applyTheme(dot.dataset.quickTheme);
        });

        const scrollTopBtn = document.querySelector('[data-footer-action="scroll-top"]');
        if (scrollTopBtn) {
            scrollTopBtn.onclick = () => window.scrollTo({ top: 0, behavior: "smooth" });
        }

        syncActiveSidebarLink();
    };

    // 6. Sự kiện toàn cục (Cuộn trang, Resize, Click ngoài Dropdown, Phím Escape)
    const bindGlobalListeners = () => {
        if (isGlobalBound) return;
        isGlobalBound = true;

        window.addEventListener(
            "scroll",
            () => {
                const header = document.getElementById("header") || document.getElementById("site-header");
                if (header) {
                    header.classList.toggle("header--scrolled", window.scrollY > 10);
                }
            },
            { passive: true },
        );

        window.addEventListener("resize", () => {
            const sidebar = document.getElementById("calendar-sidebar");
            if (window.innerWidth >= 768 && sidebar?.classList.contains("is-open")) {
                toggleSidebar(true);
            }
        });

        document.addEventListener("click", (e) => {
            document.querySelectorAll(".js-header-dropdown[open], .js-cal-dropdown[open]").forEach((dropdown) => {
                if (!dropdown.contains(e.target)) {
                    dropdown.removeAttribute("open");
                }
            });
        });

        document.addEventListener("keydown", (e) => {
            if (e.key === "Escape") {
                toggleSidebar(true);
                document.querySelectorAll(".js-header-dropdown[open], .js-cal-dropdown[open]").forEach((dropdown) => {
                    dropdown.removeAttribute("open");
                });
            }
        });
    };

    document.addEventListener("DOMContentLoaded", () => {
        bindGlobalListeners();
        initHeaderAndSidebar();
    });

    window.addEventListener("template-loaded", initHeaderAndSidebar);
})();

        if (b11 - a11 > 365) {
            const leapMonthDiff = getLeapMonthOffset(a11, timeZone);
            if (diff >= leapMonthDiff) {
                lunarMonth = diff + 10;
                if (diff === leapMonthDiff) {
                    lunarLeap = 1;
                }
            }
        }
        if (lunarMonth > 12) {
            lunarMonth = lunarMonth - 12;
        }
        if (lunarMonth >= 11 && diff < 4) {
            lunarYear -= 1;
        }
        return {
            lunarDay,
            lunarMonth,
            lunarYear,
            lunarLeap,
            isFullMonth,
            jd: dayNumber,
        };
    };

            // Thông báo nhẹ nhàng cho Screen Reader khi chuyển sang Canh Giờ mới
            const announcer = document.getElementById("clock-sr-announcer");
            if (announcer && currentHourInfo) {
                announcer.textContent = `Đã chuyển sang Giờ ${currentHourInfo.canchi}, ${currentHourInfo.isGood ? "Giờ Hoàng Đạo" : "Giờ Hắc Đạo"}`;
            }
        }
    };

    // 7. Gọi API Open-Meteo (Miễn phí, không cần Key) lấy Nhiệt độ & Giờ Mặt trời mọc/lặn theo data-lat, data-lon
    const fetchWeatherAndSun = async (lat, lon, tz) => {
        try {
            const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current_weather=true&daily=sunrise,sunset&timezone=${encodeURIComponent(tz)}`;
            const res = await fetch(url);
            if (!res.ok) return;
            const data = await res.json();

            if (data.current_weather) {
                const temp = Math.round(data.current_weather.temperature);
                const isDay = data.current_weather.is_day === 1;
                document.getElementById("clock-weather-temp").textContent = `${temp}°C`;
                document.getElementById("clock-weather-icon").textContent = isDay ? "☀️" : "🌙";
            }

            if (data.daily?.sunrise?.[0] && data.daily?.sunset?.[0]) {
                const rise = data.daily.sunrise[0].split("T")[1];
                const set = data.daily.sunset[0].split("T")[1];
                const [rH, rM] = rise.split(":").map(Number);
                const [sH, sM] = set.split(":").map(Number);
                const totalMins = sH * 60 + sM - (rH * 60 + rM);
                const dH = Math.floor(totalMins / 60);
                const dM = totalMins % 60;
                const sunEl = document.getElementById("clock-sun- times") || document.getElementById("clock-sun-times");
                if (sunEl) {
                    sunEl.textContent = `${rise} - ${set} (${dH}h ${String(dM).padStart(2, "0")}m)`;
                }
            }
        } catch (err) {
            console.warn("[clock] Không thể tải dữ liệu thời tiết, giữ giá trị mặc định:", err.message);
        }
    };

    // 8. Lắng nghe sự kiện tương tác qua dataset
    // 8.1. Nút chuyển đổi 12h / 24h
    document.querySelectorAll(".clock-hero__format-btn").forEach((btn) => {
        btn.addEventListener("click", () => {
            const format = btn.dataset.timeFormat;
            app.dataset.activeFormat = format;

            document.querySelectorAll(".clock-hero__format-btn").forEach((b) => {
                const isMatch = b === btn;
                b.classList.toggle("is-active", isMatch);
                b.setAttribute("aria-pressed", String(isMatch));
            });

            tickClock();
        });
    });

    // 8.2. Chọn Khu vực / Tỉnh thành từ Dropdown
    const locDropdown = document.getElementById("clock-loc-dropdown");
    document.querySelectorAll(".clock-hero__loc-option").forEach((opt) => {
        opt.addEventListener("click", () => {
            const { cityId, cityName, country, lat, lon, tz, utc } = opt.dataset;

            app.dataset.activeCity = cityId;
            app.dataset.timezone = tz;

            document.querySelectorAll(".clock-hero__loc-option").forEach((o) => {
                const isMatch = o === opt;
                o.classList.toggle("is-selected", isMatch);
                o.setAttribute("aria-selected", String(isMatch));
            });

            document.getElementById("clock-loc-label").textContent = `${cityName}, ${country}`;
            document.getElementById("clock-city-heading").innerHTML = `${cityName},<br />${country}`;
            document.getElementById("clock-meta-utc").textContent = utc;
            document.getElementById("schema-lat")?.setAttribute("content", lat);
            document.getElementById("schema-lon")?.setAttribute("content", lon);

            locDropdown?.removeAttribute("open");
            lastRenderedBranchIndex = -1; // Ép tính lại Can Chi theo múi giờ mới
            tickClock();
            fetchWeatherAndSun(lat, lon, tz);
        });
    });

    // 8.3. Bộ lọc Tất cả / Hoàng Đạo / Hắc Đạo
    document.querySelectorAll(".clock-zodiac-section__filter-btn").forEach((btn) => {
        btn.addEventListener("click", () => {
            document.querySelectorAll(".clock-zodiac-section__filter-btn").forEach((b) => {
                const isMatch = b === btn;
                b.classList.toggle("is-active", isMatch);
                b.setAttribute("aria-pressed", String(isMatch));
            });
            const parts = getZonedDateParts(app.dataset.timezone || "Asia/Ho_Chi_Minh");
            renderZodiacGrid(computeLunarData(parts));
        });
    });

    // Khởi chạy
    tickClock();
    setInterval(tickClock, 1000);
    fetchWeatherAndSun("10.8231", "106.6297", "Asia/Ho_Chi_Minh");
})();
