const $ = document.querySelector.bind(document);
const $$ = document.querySelectorAll.bind(document);

/**
 * Hàm tải template
 *
 * Cách dùng:
 * <div id="parent"></div>
 * <script>
 *  load("#parent", "./path-to-template.html");
 * </script>
 */
// Trong assets/js/script.js (Dòng 13 - 29)
function load(selector, path) {
    const targetEl = $(selector);
    if (!targetEl) {
        console.warn(`[load] Không tìm thấy thẻ "${selector}" trong index.html!`);
        return;
    }

    const cached = localStorage.getItem(path);
    if (cached && cached.trim() !== "") {
        targetEl.innerHTML = cached;
    }

    fetch(path)
        .then((res) => {
            if (!res.ok) {
                throw new Error(`HTTP ${res.status} - Không tìm thấy file tại đường dẫn: ${path}`);
            }
            return res.text();
        })
        .then((html) => {
            if (html.trim() === "") {
                console.warn(`[load] File "${path}" đang RỖNG (0 ký tự). Hãy kiểm tra và bấm Ctrl+S trong VS Code!`);
                return;
            }
            if (html !== cached) {
                targetEl.innerHTML = html;
                localStorage.setItem(path, html);
            }
        })
        .catch((err) => {
            console.error(`[load] Lỗi khi nạp "${path}":`, err.message);
        })
        .finally(() => {
            window.dispatchEvent(new Event("template-loaded"));
        });
}

/**
 * Hàm kiểm tra một phần tử
 * có bị ẩn bởi display: none không
 */
function isHidden(element) {
    if (!element) return true;

    if (window.getComputedStyle(element).display === "none") {
        return true;
    }

    let parent = element.parentElement;
    while (parent) {
        if (window.getComputedStyle(parent).display === "none") {
            return true;
        }
        parent = parent.parentElement;
    }

    return false;
}

/**
 * Hàm buộc một hành động phải đợi
 * sau một khoảng thời gian mới được thực thi
 */
function debounce(func, timeout = 300) {
    let timer;
    return (...args) => {
        clearTimeout(timer);
        timer = setTimeout(() => {
            func.apply(this, args);
        }, timeout);
    };
}

/**
 * Hàm tính toán vị trí arrow cho dropdown
 *
 * Cách dùng:
 * 1. Thêm class "js-dropdown-list" vào thẻ ul cấp 1
 * 2. CSS "left" cho arrow qua biến "--arrow-left-pos"
 */
const calArrowPos = debounce(() => {
    if (isHidden($(".js-dropdown-list"))) return;

    const items = $$(".js-dropdown-list > li");

    items.forEach((item) => {
        const arrowPos = item.offsetLeft + item.offsetWidth / 2;
        item.style.setProperty("--arrow-left-pos", `${arrowPos}px`);
    });
});

/**
 * Giữ active menu khi hover
 *
 * Cách dùng:
 * 1. Thêm class "js-menu-list" vào thẻ ul menu chính
 * 2. Thêm class "js-dropdown" vào class "dropdown" hiện tại
 *  nếu muốn reset lại item active khi ẩn menu
 */
window.addEventListener("template-loaded", handleActiveMenu);

function handleActiveMenu() {
    const dropdowns = $$(".js-dropdown");
    const menus = $$(".js-menu-list");
    const activeClass = "menu-column__item--active";

    const removeActive = (menu) => {
        menu.querySelector(`.${activeClass}`)?.classList.remove(activeClass);
    };

    const init = () => {
        menus.forEach((menu) => {
            const items = menu.children;
            if (!items.length) return;

            removeActive(menu);
            if (window.innerWidth > 991) items[0].classList.add(activeClass);

            Array.from(items).forEach((item) => {
                item.onmouseenter = () => {
                    if (window.innerWidth <= 991) return;
                    removeActive(menu);
                    item.classList.add(activeClass);
                };
                item.onclick = () => {
                    if (window.innerWidth > 991) return;
                    removeActive(menu);
                    item.classList.add(activeClass);
                    item.scrollIntoView();
                };
            });
        });
    };

    init();

    dropdowns.forEach((dropdown) => {
        dropdown.onmouseleave = () => init();
    });
}

/**
 * JS toggle
 *
 * Cách dùng:
 * <button class="js-toggle" toggle-target="#box">Click</button>
 * <div id="box">Content show/hide</div>
 */
window.addEventListener("template-loaded", initJsToggle);

function initJsToggle() {
    $$(".js-toggle").forEach((button) => {
        // Đọc qua dataset.toggleTarget chuẩn HTML5 (hỗ trợ lùi về toggle-target cũ nếu còn sót)
        const target = button.dataset.toggleTarget || button.getAttribute("toggle-target");
        if (!target) return;

        button.onclick = (e) => {
            e.preventDefault();
            const targetEl = $(target);
            if (!targetEl) return;

            const isHidden = targetEl.classList.contains("hide");
            requestAnimationFrame(() => {
                targetEl.classList.toggle("hide", !isHidden);
                targetEl.classList.toggle("show", isHidden);
                button.setAttribute("aria-expanded", String(isHidden));
            });
        };
    });
}

window.addEventListener("template-loaded", () => {
    const links = $$(".js-dropdown-list > li > a");

    links.forEach((link) => {
        link.onclick = () => {
            if (window.innerWidth > 991) return;
            const item = link.closest("li");
            item.classList.toggle("navbar__item--active");
        };
    });
});

// Chuyên trách duy nhất templates/sidebar.html: Ngăn kéo Mobile Drawer, Khóa cuộn Body & Active Link
(() => {
    let isSidebarGlobalBound = false;

    const toggleSidebar = (forceClose = false) => {
        const sidebar = document.getElementById("calendar-sidebar");
        const burgerBtn = document.getElementById("burger-btn");
        const backdrop = document.getElementById("sidebar-backdrop");
        if (!sidebar || !burgerBtn || !backdrop) return;

        const isOpen = sidebar.classList.contains("is-open");
        const nextState = forceClose ? false : !isOpen;

        sidebar.classList.toggle("is-open", nextState);
        burgerBtn.classList.toggle("is-active", nextState);
        backdrop.classList.toggle("is-active", nextState);
        burgerBtn.setAttribute("aria-expanded", String(nextState));

        // Khóa cuộn trang trên <body> khi mở menu Mobile
        document.body.style.overflow = nextState ? "hidden" : "";
    };

    const syncActiveSidebarLink = () => {
        const currentPage = window.location.pathname.split("/").pop() || "index.html";
        const sidebarLinks = document.querySelectorAll(".sidebar__link[data-nav-page]");

        sidebarLinks.forEach((link) => {
            const isMatch = link.dataset.navPage === currentPage;
            link.classList.toggle("sidebar__link--active", isMatch);
            if (isMatch) {
                link.setAttribute("aria-current", "page");
            } else {
                link.removeAttribute("aria-current");
            }
        });
    };

    const initSidebar = () => {
        const burgerBtn = document.getElementById("burger-btn");
        const backdrop = document.getElementById("sidebar-backdrop");

        if (burgerBtn) burgerBtn.onclick = () => toggleSidebar();
        if (backdrop) backdrop.onclick = () => toggleSidebar(true);

        syncActiveSidebarLink();
    };

    const bindSidebarGlobalListeners = () => {
        if (isSidebarGlobalBound) return;
        isSidebarGlobalBound = true;

        window.addEventListener("resize", () => {
            const sidebar = document.getElementById("calendar-sidebar");
            if (window.innerWidth >= 768 && sidebar?.classList.contains("is-open")) {
                toggleSidebar(true);
            }
        });

        document.addEventListener("keydown", (e) => {
            if (e.key === "Escape") {
                toggleSidebar(true);
            }
        });
    };

    document.addEventListener("DOMContentLoaded", () => {
        bindSidebarGlobalListeners();
        initSidebar();
    });

    window.addEventListener("template-loaded", initSidebar);
})();

        syncFooterThemeDots(activeTheme);

        footer.querySelectorAll(".footer__element-dot").forEach((dot) => {
            dot.onclick = () => {
                if (typeof window.setFengShuiTheme === "function") {
                    window.setFengShuiTheme(dot.dataset.quickTheme);
                }
            };
        });

        const scrollTopBtn = footer.querySelector('[data-footer-action="scroll-top"]');
        if (scrollTopBtn) {
            scrollTopBtn.onclick = () => window.scrollTo({ top: 0, behavior: "smooth" });
        }

        const converterBtn = footer.querySelector('[data-footer-action="open-converter"]');
        if (converterBtn) {
            converterBtn.onclick = () => {
                const headerWidgetBtn = document.getElementById("widget-trigger-btn");
                headerWidgetBtn?.click();
                headerWidgetBtn?.focus();
            };
        }

        const newsletterForm = document.getElementById("footer-newsletter-form");
        const emailInput = document.getElementById("footer-email");
        const feedbackEl = document.getElementById("footer-form-feedback");

        if (newsletterForm && emailInput && feedbackEl) {
            newsletterForm.onsubmit = (e) => {
                e.preventDefault();
                if (!emailInput.value.trim() || !emailInput.checkValidity()) {
                    feedbackEl.hidden = false;
                    feedbackEl.textContent = "Vui lòng nhập địa chỉ email hợp lệ!";
                    return;
                }
                feedbackEl.hidden = false;
                feedbackEl.textContent = "Đăng ký nhận lịch ngày tốt thành công!";
                emailInput.value = "";
            };
        }
    };

    window.addEventListener("theme-changed", (e) => {
        if (e.detail?.theme) syncFooterThemeDots(e.detail.theme);
    });

    document.addEventListener("DOMContentLoaded", initFooter);
    window.addEventListener("template-loaded", initFooter);
})();

// assets/js/clock.js
// Chuyên trách duy nhất Trang Đồng hồ (clock.html): Đồng hồ TimeSpot, Bát Tự Tứ Trụ, 12 Canh Giờ & Thời tiết
(() => {
    let clockInterval = null;
    let lastRenderedBranchIndex = -1;
    let selectedBranchIndex = -1;

    const getZonedDateParts = (timeZone) => {
        const now = new Date();
        const formatter = new Intl.DateTimeFormat("en-US", {
            timeZone,
            year: "numeric",
            month: "2-digit",
            day: "2-digit",
            hour: "2-digit",
            minute: "2-digit",
            second: "2-digit",
            hour12: false,
        });
        const parts = Object.fromEntries(formatter.formatToParts(now).map((p) => [p.type, p.value]));
        return {
            year: Number(parts.year),
            month: Number(parts.month),
            day: Number(parts.day),
            hour: Number(parts.hour) % 24,
            minute: Number(parts.minute),
            second: Number(parts.second),
        };
    };

    const updateSelectedHourDetail = (hourObj, solarTerm) => {
        if (!hourObj) return;
        const nameEl = document.getElementById("detail-selected-hour-name");
        const napamEl = document.getElementById("detail-selected-hour-napam");
        const termEl = document.getElementById("detail-selected-solarterm");

        if (nameEl) nameEl.textContent = `Giờ ${hourObj.canchi} (${hourObj.range})`;
        if (napamEl) napamEl.textContent = hourObj.napAm;
        if (termEl) termEl.textContent = solarTerm;
    };

    const renderZodiacGrid = (dayInfo) => {
        const grid = document.getElementById("clock-zodiac-grid");
        if (!grid) return;

        const activeFilter =
            document.querySelector(".clock-zodiac-section__filter-btn.is-active")?.dataset.hourFilter || "all";

        grid.innerHTML = dayInfo.hours12
            .map((h) => {
                const isCurrent = h.index === dayInfo.currentBranchIdx;
                const isSelected = h.index === selectedBranchIndex;
                const fortuneKey = h.isGood ? "good" : "bad";
                const fortuneLabel = h.isGood ? "Hoàng Đạo" : "Hắc Đạo";
                const fortuneIcon = h.isGood ? "☀️" : "🌙";
                const isDimmed = activeFilter !== "all" && activeFilter !== fortuneKey;

                const focusTargetIdx = selectedBranchIndex !== -1 ? selectedBranchIndex : dayInfo.currentBranchIdx;
                const tabIdx = h.index === focusTargetIdx ? "0" : "-1";

                return `
                    <button
                        type="button"
                        class="clock-zodiac-card ${isCurrent ? "is-current" : ""} ${isSelected ? "is-selected" : ""} ${isDimmed ? "is-dimmed" : ""}"
                        tabindex="${tabIdx}"
                        ${isCurrent ? 'aria-current="time"' : ""}
                        aria-label="Giờ ${h.canchi}, từ ${h.range}, ${fortuneLabel}, mệnh ${h.napAm}"
                        title="Giờ ${h.canchi} (${h.range}) • ${fortuneLabel} • Mệnh ${h.napAm}"
                        data-branch-index="${h.index}"
                        data-branch="${h.key}"
                        data-canchi="${h.canchi}"
                        data-range="${h.range}"
                        data-napam="${h.napAm}"
                        data-fortune="${fortuneKey}"
                    >
                        <div class="clock-zodiac-card__top" aria-hidden="true">
                            <span class="clock-zodiac-card__name">Giờ ${h.vn}</span>
                            <span class="clock-zodiac-card__utc">${h.range}</span>
                        </div>
                        <div class="clock-zodiac-card__bottom" aria-hidden="true">
                            <strong class="clock-zodiac-card__canchi">${h.canchi}</strong>
                            <span class="clock-zodiac-card__fortune">
                                <span>${fortuneIcon}</span> ${fortuneLabel}
                            </span>
                        </div>
                        <img
                            src="./assets/icons/zodiac/${h.icon}"
                            alt=""
                            aria-hidden="true"
                            class="clock-zodiac-card__img"
                        />
                    </button>
                `;
            })
            .join("");

        const cards = Array.from(grid.querySelectorAll(".clock-zodiac-card"));
        cards.forEach((card, idx) => {
            card.onclick = () => {
                selectedBranchIndex = Number(card.dataset.branchIndex);
                updateSelectedHourDetail(dayInfo.hours12[selectedBranchIndex], dayInfo.solarTerm);
                renderZodiacGrid(dayInfo);
            };

            card.onkeydown = (e) => {
                let nextIdx = -1;
                if (e.key === "ArrowRight" || e.key === "ArrowDown") nextIdx = (idx + 1) % cards.length;
                if (e.key === "ArrowLeft" || e.key === "ArrowUp") nextIdx = (idx - 1 + cards.length) % cards.length;

                if (nextIdx !== -1) {
                    e.preventDefault();
                    selectedBranchIndex = Number(cards[nextIdx].dataset.branchIndex);
                    updateSelectedHourDetail(dayInfo.hours12[selectedBranchIndex], dayInfo.solarTerm);
                    renderZodiacGrid(dayInfo);
                    grid.querySelector(`[data-branch-index="${selectedBranchIndex}"]`)?.focus();
                }
            };
        });
    };

    const fetchWeatherAndSun = async (lat, lon, tz) => {
        try {
            const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current_weather=true&daily=sunrise,sunset&timezone=${encodeURIComponent(tz)}`;
            const res = await fetch(url);
            if (!res.ok) return;
            const data = await res.json();

            if (data.current_weather) {
                const temp = Math.round(data.current_weather.temperature);
                const isDay = data.current_weather.is_day === 1;
                const tempEl = document.getElementById("clock-weather-temp");
                const iconEl = document.getElementById("clock-weather-icon");
                if (tempEl) tempEl.textContent = `${temp}°C`;
                if (iconEl) iconEl.textContent = isDay ? "☀️" : "🌙";
            }

            if (data.daily?.sunrise?.[0] && data.daily?.sunset?.[0]) {
                const rise = data.daily.sunrise[0].split("T")[1];
                const set = data.daily.sunset[0].split("T")[1];
                const [rH, rM] = rise.split(":").map(Number);
                const [sH, sM] = set.split(":").map(Number);
                const totalMins = sH * 60 + sM - (rH * 60 + rM);
                const dH = Math.floor(totalMins / 60);
                const dM = totalMins % 60;
                const sunEl = document.getElementById("clock-sun-times") || document.getElementById("clock-sun- times");
                if (sunEl) {
                    sunEl.textContent = `${rise} - ${set} (${dH}h ${String(dM).padStart(2, "0")}m)`;
                }
            }
        } catch (err) {
            // Giữ giá trị mặc định nếu không có kết nối mạng
        }
    };

    const initClockPage = () => {
        const rootEl = document.getElementById("clock-calendar-root");

            const format = app.dataset.activeFormat || "24";
            const parts = getZonedDateParts(tz);
            const pad = (n) => String(n).padStart(2, "0");

            let displayHour = parts.hour;
            const ampmEl = document.getElementById("hero-clock-ampm");

            if (format === "12") {

            const secondsEl = app.querySelector('[data-clock-part="seconds"]');

            if (minutesEl) minutesEl.textContent = pad(parts.minute);
            if (secondsEl) secondsEl.textContent = pad(parts.second);
            if (timeTag) {
                timeTag.setAttribute("datetime", `${pad(parts.hour)}:${pad(parts.minute)}:${pad(parts.second)}`);
            }

            const currentBranchIdx = window.LunarEngine.getBranchIndexFromHour(parts.hour);
            if (currentBranchIdx !== lastRenderedBranchIndex) {
                lastRenderedBranchIndex = currentBranchIdx;
                const dayInfo = window.LunarEngine.getFullDayInfo(parts.day, parts.month, parts.year, parts.hour);

                    statusText.textContent = `Đang là Giờ ${currentHourInfo.canchi} • ${currentHourInfo.isGood ? "Hoàng Đạo" : "Hắc Đạo"}`;

                    timeZone: tz,
                    weekday: "long",
                    day: "2-digit",
                    month: "2-digit",
                    year: "numeric",

                    document.getElementById("pillar-hour-fortune").textContent = currentHourInfo.isGood

                renderZodiacGrid(dayInfo);

                if (announcer && currentHourInfo) {

        lastRenderedBranchIndex = -1;
        tickClock();

        C1 = C1 + 0.0104 * Math.sin(dr * 2 * F) - 0.0051 * Math.sin(dr * (M + Mpr));
        C1 = C1 - 0.0074 * Math.sin(dr * (M - Mpr)) + 0.0004 * Math.sin(dr * (2 * F + M));
        C1 = C1 - 0.0004 * Math.sin(dr * (2 * F - M)) - 0.0006 * Math.sin(dr * (2 * F + Mpr));
        C1 = C1 + 0.001 * Math.sin(dr * (2 * F - Mpr)) + 0.0005 * Math.sin(dr * (2 * Mpr + M));
        let deltat;
        if (T < -11) {
            deltat = 0.001 + 0.000839 * T + 0.0002261 * T2 - 0.00000845 * T3 - 0.000000081 * T * T3;
        } else {
            deltat = -0.000278 + 0.000265 * T + 0.000262 * T2;
        }
        const JdNew = Jd1 + C1 - deltat;
        return INT(JdNew + 0.5 + timeZone / 24);
    };

    const getSunLongitude = (jdn, timeZone = 7) => {
        const T = (jdn - 2451545.5 - timeZone / 24) / 36525;
        const T2 = T * T;
        const dr = Math.PI / 180;
        const M = 357.5291 + 35999.0503 * T - 0.0001559 * T2 - 0.00000048 * T * T2;
        const L0 = 280.46645 + 36000.76983 * T + 0.0003032 * T2;
        let DL = (1.9146 - 0.004817 * T - 0.000014 * T2) * Math.sin(dr * M);
        DL = DL + (0.019993 - 0.000101 * T) * Math.sin(dr * 2 * M) + 0.00029 * Math.sin(dr * 3 * M);
        let L = L0 + DL;
        L = L * dr;
        L = L - Math.PI * 2 * INT(L / (Math.PI * 2));
        return INT((L / Math.PI) * 6);
    };

    const getLunarMonth11 = (yy, timeZone = 7) => {
        const off = jdFromDate(31, 12, yy) - 2415021;
        const k = INT(off / 29.530588853);
        let nm = getNewMoonDay(k, timeZone);
        const sunLong = getSunLongitude(nm, timeZone);
        if (sunLong >= 9) {
            nm = getNewMoonDay(k - 1, timeZone);
        }
        return nm;
    };

    const getLeapMonthOffset = (a11, timeZone = 7) => {
        const k = INT((a11 - 2415021.076998695) / 29.530588853 + 0.5);
        let last = 0;
        let i = 1;
        let arc = getSunLongitude(getNewMoonDay(k + i, timeZone), timeZone);
        do {
            last = arc;
            i++;
            arc = getSunLongitude(getNewMoonDay(k + i, timeZone), timeZone);
        } while (arc !== last && i < 14);
        return i - 1;
    };

    const convertSolar2Lunar = (dd, mm, yy, timeZone = 7) => {
        const dayNumber = jdFromDate(dd, mm, yy);
        const k = INT((dayNumber - 2415021.076998695) / 29.530588853);
        let monthStart = getNewMoonDay(k + 1, timeZone);
        if (monthStart > dayNumber) {
            monthStart = getNewMoonDay(k, timeZone);
        }
        let nextMonthStart = getNewMoonDay(k + 1, timeZone);
        if (nextMonthStart <= monthStart) {
            nextMonthStart = getNewMoonDay(k + 2, timeZone);
        }
        const isFullMonth = nextMonthStart - monthStart === 30;

        let a11 = getLunarMonth11(yy, timeZone);
        let b11 = a11;
        let lunarYear;
        if (a11 >= monthStart) {
            lunarYear = yy;
            a11 = getLunarMonth11(yy - 1, timeZone);
        } else {
            lunarYear = yy + 1;
            b11 = getLunarMonth11(yy + 1, timeZone);
        }
        const lunarDay = dayNumber - monthStart + 1;
        const diff = INT((monthStart - a11) / 29);
        let lunarLeap = 0;
        let lunarMonth = diff + 11;
        if (b11 - a11 > 365) {
            const leapMonthDiff = getLeapMonthOffset(a11, timeZone);
            if (diff >= leapMonthDiff) {
                lunarMonth = diff + 10;
                if (diff === leapMonthDiff) {
                    lunarLeap = 1;
                }
            }
        }
        if (lunarMonth > 12) {
            lunarMonth = lunarMonth - 12;
        }
        if (lunarMonth >= 11 && diff < 4) {
            lunarYear -= 1;
        }
        return { lunarDay, lunarMonth, lunarYear, lunarLeap, isFullMonth, jd: dayNumber };
    };

    const convertLunar2Solar = (lunarDay, lunarMonth, lunarYear, lunarLeap = 0, timeZone = 7) => {
        let a11, b11;
        if (lunarMonth < 11) {
            a11 = getLunarMonth11(lunarYear - 1, timeZone);
            b11 = getLunarMonth11(lunarYear, timeZone);
        } else {
            a11 = getLunarMonth11(lunarYear, timeZone);
            b11 = getLunarMonth11(lunarYear + 1, timeZone);
        }
        const k = INT(0.5 + (a11 - 2415021.076998695) / 29.530588853);
        let off = lunarMonth - 11;
        if (off < 0) off += 12;
        if (b11 - a11 > 365) {
            const leapOff = getLeapMonthOffset(a11, timeZone);
            let leapMonth = leapOff - 2;
            if (leapMonth < 0) leapMonth += 12;
            if (lunarLeap !== 0 && lunarMonth !== leapMonth) {
                return { sDay: 0, sMonth: 0, sYear: 0 };
            } else if (lunarLeap !== 0 || off >= leapOff) {
                off += 1;
            }
        }
        const monthStart = getNewMoonDay(k + off, timeZone);
        return jdToDate(monthStart + lunarDay - 1);
    };

    const getNapAm = (stemIdx, branchIdx) => {
        const idx = (((6 * stemIdx - 5 * branchIdx) % 60) + 60) % 60;
        return NAP_AM_LIST[Math.floor(idx / 2)] || "Bạch Lạp Kim";
    };

    const resolveElementInfo = (napAmStr) => {

        if (text.endsWith("Mộc")) return { icon: "wood.svg", label: "Hành Mộc • Sinh trưởng" };
        if (text.endsWith("Thủy")) return { icon: "water.svg", label: "Hành Thủy • Linh hoạt" };
        if (text.endsWith("Hỏa")) return { icon: "fire.svg", label: "Hành Hỏa • Nhiệt huyết" };
        return { icon: "earth.svg", label: "Hành Thổ • Vững chãi" };
    };

    const getDayAuspiciousStatus = (lunarMonth, dayChiIndex) => {
        const startChi = ((lunarMonth - 1) * 2) % 12;
        const offset = (dayChiIndex - startChi + 12) % 12;
        if ([0, 1, 4, 7].includes(offset)) return "good";
        if ([2, 3, 6, 8].includes(offset)) return "bad";
        return "";
    };

    const getSolarTermName = (dd, mm) => {

    };

    const getHolidayBadge = (sDay, sMonth, lDay, lMonth, lLeap) => {
        const solarKey = `${sDay}/${sMonth}`;
        const lunarKey = `${lDay}/${lMonth}`;

        if (lDay === 1) return { badgeText: "Mùng 1", isHoliday: true };
        if (lDay === 15) return { badgeText: "Rằm", isHoliday: true };
        return { badgeText: "", isHoliday: false };
    };

        const solarMonthEnLabel = enMonthFormatter.format(new Date(sYear, sMonth - 1, 1));
        const solarMonthVnLabel = VI_TRADITIONAL_SOLAR_MONTHS[sMonth] || `THÁNG ${sMonth}`;

        const baseLunarMonthName = VI_TRADITIONAL_LUNAR_MONTHS[lunarMonth] || `THÁNG ${lunarMonth}`;
        const lunarMonthFullLabel = `${baseLunarMonthName}${lunarLeap ? " NHUẬN" : ""}${isFullMonth ? " (Đ)" : " (T)"}`;

        const fortuneStatus = getDayAuspiciousStatus(lunarMonth, dayChiIdx);
        const holidayInfo = getHolidayBadge(sDay, sMonth, lunarDay, lunarMonth, lunarLeap);
        const solarTerm = getSolarTermName(sDay, sMonth);
        const trucObj = TRUC_LIST[(dayChiIdx - monthChiIdx + 12) % 12] || TRUC_LIST[0];
        const ageHarmony = AGE_HARMONY_MAP[dayChiIdx] || AGE_HARMONY_MAP[0];

        const goodHoursShort = [];

                    icon: slot.icon,
                });
            } else {
                badHoursShort.push(slot.shortRange);
            }
        });

            lunarMonth,
            lunarYear,

            monthChiIdx,
            dayChiIdx,
            yearCanChi,
            monthCanChi,

        };
    };

    window.LunarEngine = {
        THIEN_CAN,
        DIA_CHI,
        ZODIAC_FILES,
        convertSolar2Lunar,
        convertLunar2Solar,
        getFullDayInfo,
    };
})();

// assets/js/week-calendar.js
(() => {
    const initWeekCalendar = () => {
        const rootEl = document.getElementById("week-calendar-root");
        const app = document.getElementById("week-cal-app");
        if (!app) return;

        if (rootEl) rootEl.setAttribute("aria-busy", "false");
        if (app.dataset.bound === "true") return;
        app.dataset.bound = "true";

        const cards = Array.from(app.querySelectorAll(".week-cal__card"));

        const selectDayCard = (targetCard, shouldFocus = false) => {
            if (!targetCard) return;

            // 1. Lấy toàn bộ dữ liệu từ data-* của thẻ ngày thông qua dataset
            const {
                date,
                solar,
                weekday,
                lunar,
                canchi,
                napam,
                truc,
                term,
                zodiac,
                fortune,
                fortuneLabel,
                goodHours,
                do: doText,
                dont: dontText,
            } = targetCard.dataset;

            // 2. Cập nhật Roving Tabindex & aria-current="date"
            cards.forEach((card) => {
                const isSelected = card === targetCard;
                card.classList.toggle("week-cal__card--active", isSelected);
                card.setAttribute("tabindex", isSelected ? "0" : "-1");
                if (isSelected) card.setAttribute("aria-current", "date");
                else card.removeAttribute("aria-current");
            });

            // 3. Đổ dữ liệu sang Thẻ Hero bên trái (aria-live="polite" sẽ tự động đọc cho Screen Reader)
            app.dataset.activeDate = date;
            document.getElementById("hero-solar-display").textContent = solar;
            document.getElementById("hero-solar-display").setAttribute("datetime", date);
            document.getElementById("hero-weekday-display").textContent = weekday;
            document.getElementById("hero-lunar-display").innerHTML =
                `Ngày ${lunar} Âm lịch • <strong>${canchi}</strong> (${napam})`;
            document.getElementById("hero-day-fortune").textContent = fortuneLabel;
            document.getElementById("hero-day-fortune").dataset.fortune = fortune;
            document.getElementById("hero-good-hours").textContent = goodHours;
            document.getElementById("hero-truc-term").textContent = `Tiết khí: ${term} • Trực: ${truc}`;
            document.getElementById("hero-do-text").textContent = doText;
            document.getElementById("hero-dont-text").textContent = dontText;
            document.getElementById("hero-detail-link").href = `./index.html?date=${date}`;
            document.getElementById("hero-zodiac-img").src = `./assets/icons/zodiac/${zodiac}.svg`;

            if (shouldFocus) targetCard.focus();
        };

        // 4. Lắng nghe Click & Điều hướng phím mũi tên chuẩn a11y
        cards.forEach((card, idx) => {
            card.addEventListener("click", () => selectDayCard(card));

            card.addEventListener("keydown", (e) => {
                let nextIdx = -1;
                if (e.key === "ArrowRight" || e.key === "ArrowDown") nextIdx = (idx + 1) % cards.length;
                if (e.key === "ArrowLeft" || e.key === "ArrowUp") nextIdx = (idx - 1 + cards.length) % cards.length;

                if (nextIdx !== -1) {
                    e.preventDefault();
                    selectDayCard(cards[nextIdx], true);
                }
            });
        });
    };

    document.addEventListener("DOMContentLoaded", initWeekCalendar);
    window.addEventListener("template-loaded", initWeekCalendar);
})();
